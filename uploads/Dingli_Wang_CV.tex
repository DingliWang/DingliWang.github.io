\section{Research Papers}

\needspace{6\baselineskip}
\begin{tabularx}{\textwidth}{@{}Xr@{}}
\textbf{Robust Replacement of Human Decision Makers under Distribution Shift}
& \textit{Job Market Paper}
\end{tabularx}
{\textit{Revise and resubmit, Journal of Econometrics}}\par
{\small
I study when an algorithm should replace a human decision maker if the deployment
environment differs from the training environment. A distributionally robust ROC
framework compares classification performance across a set of possible deployment
distributions and examines the consequences of uncertainty for replacement decisions.
}\par

\vspace{6pt}

\paper
{Median-Anchored Adjustment of Joint VaR--ES Forecasts
 {\small\textnormal{with Tae-Hwy Lee}}}
{\textit{Under review, Journal of Business \& Economic Statistics}}
{A one-parameter adjustment to existing tail-risk forecasts that rescales their
distances from the fitted median, preserves VaR--ES ordering, and has a closed-form
spread-weighted quantile estimator.}

\vspace{6pt}

\paper
{Adversarial Density Forecast for Macro-financial Risks
 {\small\textnormal{with Tae-Hwy Lee}}}
{}
{An anchored conditional Wasserstein generative model for density forecasting in
short macroeconomic samples, with theory and evidence on distributional accuracy
and sensitivity to training randomness.}

\vspace{6pt}

\paper
{When Should Forecasts Be Equally Weighted? Signal Recovery and Estimation Costs}
 %{\small\textnormal{with Xinran Liu}}}
{}
{A comparison of equal and estimated forecast weights that separates recovered
predictive signal from estimation costs, accounts for tuning and component estimation,
and uses past out-of-sample losses to guide the choice.}
