% ===================== JOB MARKET PAPER =====================
\section{Job Market Paper}
\textbf{``Robust Replacement of Human Decision Makers under Covariate Shift.''} \me. \\
\textit{Under review, Journal of Econometrics.}

\vspace{0.6ex}
\noindent\hangindent=1.2em
\textit{Abstract.}\, I study when a machine-learning algorithm should replace a human decision maker once the deployment environment differs from the environment in which it was trained. Modeling this distribution shift through a conditional Wasserstein ambiguity set, I characterize a distributionally robust ROC frontier that bounds achievable performance under the shift, and I derive a finite-sample welfare-regret bound that governs the replacement decision and yields a feasible rule for when automation dominates human judgment.

% ===================== WORKING PAPERS =====================
\section{Working Papers}
\begin{enumerate}
  \item \textbf{``Inferential Theory for Factor-Augmented Quantile Regression under Weak Loadings.''} \me. \textit{Under review, Journal of Econometrics.}\\
  {\small $\sqrt{T}$-rate inference for factor-augmented quantile regression under weak loadings, with a Procrustes-aligned split-panel jackknife for bias correction; U.S.\ inflation-at-risk application.}

  \item \textbf{``Conditional Distributionally Robust Estimation with Wasserstein Ambiguity.''} \me. \textit{Under review, Journal of Econometrics.}\\
  {\small Distributionally robust M-estimation under a conditional Wasserstein ambiguity set (fix $P_X$, perturb $Y\mid X$), with pilot-anchored regularization; U.S.\ Growth-at-Risk application.}

  \item \textbf{``Forecast Combination with AI-Generated Narrative States: Bias, Correction, and Inference.''} \me\ and Xinran Liu. \textit{Under review, Journal of Econometrics.}\\
  {\small Plug-in inference fails at first order when the weight-driving ``narrative state'' is estimated from text; a block cross-fitting correction restores valid inference; Federal Reserve communication application.}

  \item \textbf{``Generated Covariates and Inference for Heterogeneous Treatment Effects under a Scalar Factor Structure.''} Xinran Liu and \me. \textit{Under review, Journal of Econometrics.}\\
  {\small Plug-in CATE estimators inherit a contamination term that Neyman orthogonality does not remove when conditioning covariates are generated; corrected via cross-fitting at the covariate-generation stage.}

  \item \textbf{``Localizing Distributional Granger Predictability by Wasserstein Projection.''} Xinran Liu and \me. \textit{Under review, Journal of Business \& Economic Statistics.}\\
  {\small A Wasserstein-projection test localizing whether a predictor enters the location versus the tail of the conditional distribution; weighted chi-square limit theory.}

  \item \textbf{``When Should Forecasts Be Equally Weighted? A Signal--Cost Diagnostic for Forecast Combination.''} \me. \textit{Working paper.}\\
  {\small A signal--cost diagnostic for when adaptive forecast combination dominates simple equal weights; FRED-MD rolling-forecast evidence.}
\end{enumerate}
